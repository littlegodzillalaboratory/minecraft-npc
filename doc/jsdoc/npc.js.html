  async _doSkill(skill, opts, validation, _successMessage, failureMessage) {
    if (validation()) {
      try {
        const result = await skill.do(opts);
        return result?.message
          ? { ...result, status: result.status || "success" }
          : "success";
      } catch (err) {
        bag.logStepItemError(`${failureMessage}: ${err.message}`);
        return {
          status: "failed",
          message: err.message,
        };
      }
    }

    bag.logStepItemError(failureMessage);
    return "failed";
  }

  /**
   * Execute a read-only domain skill and retain the value for action-level
   * orchestration. Query failures are returned instead of speaking directly.
   */
  async _doQuerySkill(
    skill,
    opts,
    validation,
    _successMessage,
    failureMessage,
  ) {
    if (!validation()) {
      bag.logStepItemError(failureMessage);
      return { status: "failed", message: failureMessage };
    }

    try {
      const value = await skill.do(opts);
      return { status: "success", value: value };
    } catch (err) {
      bag.logStepItemError(`${failureMessage}: ${err.message}`);
      return { status: "failed", message: err.message };
    }
  }
